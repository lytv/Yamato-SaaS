export const phase2_business_logic = `
// Core Business Logic cần implement:

1. Plan → Product Filtering:
   SELECT product.* 
   FROM plan_detail 
   JOIN product ON product.product_code = plan_detail.product_code
   WHERE plan_detail.plan_id = ? AND plan_detail.owner_id = ?

2. Product → Production Step Filtering:
   SELECT production_step.*, production_step_detail.*
   FROM production_step_detail
   JOIN production_step ON production_step.id = production_step_detail.production_step_id
   WHERE production_step_detail.product_id = ? AND production_step_detail.owner_id = ?
   ORDER BY production_step_detail.sequence_number

3. Quantity Validation:
   - Check ordered_quantity <= plan_detail.planned_quantity
   - Multiple order details có thể reference cùng plan/product

4. Denormalization:
   - Auto-fill planCode, planName từ plan table
   - Auto-fill productCode, productName từ product table  
   - Auto-fill stepCode, stepName từ production_step table
`;
